Pseudocode — VALIDATE Button (Core Logic)

-- Called when user clicks the Validate button --

FUNCTION validateForm():

    -- Run EVERY validator (no short-circuit so ALL errors show at once) --
    results = [
        validateFirstname(),
        validateMiddleinit(),
        validateLastname(),
        validateGender(),
        validateDob(),
        validateSsn(),
        validateAddr1(),
        validateAddr2(),
        validateCity(),
        validateState(),
        validateZip(),
        validateEmail(),
        validatePhone(),
        validateHistory(),   -- also validates symptoms textarea
        validateVaccinated(),
        validateInsurance(),
        validateUserid(),
        validatePassword(),
        validatePassword2()
    ]

    IF ALL results are TRUE:
        Show Submit button   → PASS
        RETURN TRUE
    ELSE:
        Keep Submit hidden   → FAIL
        RETURN FALSE

FUNCTION validateFirstname():
    value = trim(firstname field)
    IF empty                        → showError, RETURN FALSE
    IF not matching [A-Za-z'-]{1,30} → showError, RETURN FALSE
    clearError, RETURN TRUE

FUNCTION validateMiddleinit():
    value = trim(middleinit field)
    IF empty                        → clearError, RETURN TRUE  (optional)
    IF not matching [A-Za-z]{1}      → showError, RETURN FALSE
    clearError, RETURN TRUE

FUNCTION validateLastname():
    -- Same logic as validateFirstname --

FUNCTION validateGender():
    IF no radio button checked       → showError, RETURN FALSE
    clearError, RETURN TRUE

FUNCTION validateDob():
    value = date picker value
    IF empty                         → showError, RETURN FALSE
    Parse year, month, day from value
    Build Date object from parsed parts
    today    = today's date at midnight
    minDate  = today minus 120 years
    IF selected date > today         → showError "Cannot be in the future", RETURN FALSE
    IF selected date < minDate       → showError "More than 120 years ago", RETURN FALSE
    clearError, RETURN TRUE

FUNCTION validateSsn():
    Auto-format input (insert dashes after digits 3 and 5)
    value = ssn field value
    IF empty                         → showError, RETURN FALSE
    IF not matching ###-##-####       → showError, RETURN FALSE
    clearError, RETURN TRUE

FUNCTION validateAddr1():
    value = trim(addr1 field)
    IF empty                         → showError, RETURN FALSE
    IF length < 2 OR > 30            → showError, RETURN FALSE
    IF contains disallowed characters → showError, RETURN FALSE
    clearError, RETURN TRUE

FUNCTION validateAddr2():
    value = trim(addr2 field)
    IF empty                         → clearError, RETURN TRUE  (optional)
    IF length < 2 OR > 30            → showError, RETURN FALSE
    IF contains disallowed characters → showError, RETURN FALSE
    clearError, RETURN TRUE

FUNCTION validateCity():
    value = trim(city field)
    IF empty                         → showError, RETURN FALSE
    IF length < 2 OR > 30            → showError, RETURN FALSE
    IF contains digits or symbols     → showError, RETURN FALSE
    clearError, RETURN TRUE

FUNCTION validateState():
    IF dropdown value is empty        → showError, RETURN FALSE
    clearError, RETURN TRUE

FUNCTION validateZip():
    value = trim(zip field)
    IF empty                         → showError, RETURN FALSE
    IF not matching exactly 5 digits  → showError, RETURN FALSE
    clearError, RETURN TRUE

FUNCTION validateEmail():
    Auto-lowercase the email field value
    value = trim(email field)
    IF empty                         → showError, RETURN FALSE
    IF not matching name@domain.tld   → showError, RETURN FALSE
    clearError, RETURN TRUE

FUNCTION validatePhone():
    Auto-format input as ###-###-####
    value = phone field value
    IF empty                         → showError, RETURN FALSE
    IF not matching ###-###-####      → showError, RETURN FALSE
    clearError, RETURN TRUE

FUNCTION validateHistory():
    -- Checkboxes are optional; delegates to validateSymptoms --
    RETURN validateSymptoms()

FUNCTION validateSymptoms():
    value = trim(textarea)
    IF empty                         → RETURN TRUE  (optional)
    IF length > 500                  → showError, RETURN FALSE
    IF contains HTML tags            → showError, RETURN FALSE
    RETURN TRUE

FUNCTION validateVaccinated() / validateInsurance():
    IF no radio button checked        → showError, RETURN FALSE
    clearError, RETURN TRUE

FUNCTION validateUserid():
    Auto-lowercase the userid field value
    value = trim(userid field)
    IF empty                         → showError, RETURN FALSE
    IF starts with a digit            → showError, RETURN FALSE
    IF contains spaces               → showError, RETURN FALSE
    IF not matching [a-z][a-z0-9_-]{4,29} → showError, RETURN FALSE
    clearError, RETURN TRUE

FUNCTION validatePassword():
    pw      = password field value
    userid  = userid field value (lowercase)
    fname   = firstname field value
    lname   = lastname field value
    IF empty                              → showError, RETURN FALSE
    IF length < 8                         → showError, RETURN FALSE
    IF no uppercase letter               → showError, RETURN FALSE
    IF no lowercase letter               → showError, RETURN FALSE
    IF no digit                          → showError, RETURN FALSE
    IF password (lowercase) equals userid
       OR contains userid, fname, or lname → showError, RETURN FALSE
    clearError
    IF confirm field is not empty → re-run validatePassword2()
    RETURN TRUE

FUNCTION validatePassword2():
    pw  = password field value
    pw2 = confirm password field value
    IF pw2 is empty                   → showError, RETURN FALSE
    IF pw ≠ pw2                       → showError, RETURN FALSE
    clearError, RETURN TRUE

Pseudocode — Password Strength Meter

-- Runs on every keystroke in the password field --

FUNCTION updateStrength():
    pw    = password field value
    score = 0

    IF pw is empty → hide bar, clear label, RETURN
    Show the strength bar

    IF length ≥ 8         → score + 1
    IF length ≥ 12        → score + 1
    IF has uppercase      → score + 1
    IF has lowercase      → score + 1
    IF has digit          → score + 1
    IF has special char   → score + 1

    IF score ≤ 2  → bar = 33%,  color = red,    label = "Weak"
    IF score ≤ 4  → bar = 66%,  color = orange, label = "Medium"
    ELSE          → bar = 100%, color = green,  label = "Strong"
